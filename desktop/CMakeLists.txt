cmake_minimum_required(VERSION 3.22)
project(WhippetDesktop VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets Network Test)
qt_standard_project_setup()
qt_add_executable(whippet-desktop WIN32 MACOSX_BUNDLE src/main.cpp src/desktop.cpp src/desktop.h src/rpc.cpp src/rpc.h src/model.h vendor/qrcodegen.cpp vendor/qrcodegen.hpp)
target_link_libraries(whippet-desktop PRIVATE Qt6::Widgets Qt6::Network)
target_include_directories(whippet-desktop PRIVATE vendor)
set_target_properties(whippet-desktop PROPERTIES OUTPUT_NAME Whippet MACOSX_BUNDLE_BUNDLE_NAME Whippet MACOSX_BUNDLE_GUI_IDENTIFIER org.whippet.desktop MACOSX_BUNDLE_SHORT_VERSION_STRING ${PROJECT_VERSION})
if(WIN32)
  configure_file(packaging/icon.rc.in icon.rc @ONLY)
  target_sources(whippet-desktop PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/icon.rc")
elseif(APPLE)
  set_source_files_properties(../src/qt/res/icons/whippet.icns PROPERTIES MACOSX_PACKAGE_LOCATION Resources)
  target_sources(whippet-desktop PRIVATE ../src/qt/res/icons/whippet.icns)
  set_target_properties(whippet-desktop PROPERTIES MACOSX_BUNDLE_ICON_FILE whippet.icns)
endif()
include(CTest)
if(BUILD_TESTING)
  qt_add_executable(desktop-tests tests/tests.cpp src/rpc.cpp src/rpc.h src/model.h)
  target_include_directories(desktop-tests PRIVATE src)
  target_link_libraries(desktop-tests PRIVATE Qt6::Test Qt6::Network)
  add_test(NAME desktop-tests COMMAND desktop-tests)
  qt_add_executable(desktop-ui-tests tests/ui_tests.cpp src/desktop.cpp src/desktop.h src/rpc.cpp src/rpc.h src/model.h vendor/qrcodegen.cpp vendor/qrcodegen.hpp)
  target_include_directories(desktop-ui-tests PRIVATE src vendor)
  target_link_libraries(desktop-ui-tests PRIVATE Qt6::Test Qt6::Widgets Qt6::Network)
  add_test(NAME desktop-ui-tests COMMAND desktop-ui-tests)
  set_tests_properties(desktop-tests desktop-ui-tests PROPERTIES ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:$<TARGET_FILE_DIR:Qt6::Core>" RUN_SERIAL TRUE)
endif()
include(GNUInstallDirs)
install(TARGETS whippet-desktop BUNDLE DESTINATION . RUNTIME DESTINATION bin)
set(WHIPPET_NODE_DIR "" CACHE PATH "Directory containing verified whippetd and whippet-cli binaries")
if(NOT WHIPPET_NODE_DIR OR NOT EXISTS "${WHIPPET_NODE_DIR}/whippetd${CMAKE_EXECUTABLE_SUFFIX}")
  message(FATAL_ERROR "Provide WHIPPET_NODE_DIR with the verified Whippet 1.3.1 binaries (run packaging/fetch_node.py).")
endif()
add_custom_command(TARGET whippet-desktop POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_FILE_DIR:whippet-desktop>/node"
  COMMAND ${CMAKE_COMMAND} -E copy_if_different "${WHIPPET_NODE_DIR}/whippetd${CMAKE_EXECUTABLE_SUFFIX}" "$<TARGET_FILE_DIR:whippet-desktop>/node/whippetd${CMAKE_EXECUTABLE_SUFFIX}")
if(BUILD_TESTING)
  add_custom_command(TARGET desktop-ui-tests POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_FILE_DIR:desktop-ui-tests>/node"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${WHIPPET_NODE_DIR}/whippetd${CMAKE_EXECUTABLE_SUFFIX}" "$<TARGET_FILE_DIR:desktop-ui-tests>/node/whippetd${CMAKE_EXECUTABLE_SUFFIX}")
endif()
if(APPLE)
  install(PROGRAMS "${WHIPPET_NODE_DIR}/whippetd" "${WHIPPET_NODE_DIR}/whippet-cli" DESTINATION Whippet.app/Contents/MacOS/node)
elseif(WIN32)
  install(PROGRAMS "${WHIPPET_NODE_DIR}/whippetd${CMAKE_EXECUTABLE_SUFFIX}" "${WHIPPET_NODE_DIR}/whippet-cli${CMAKE_EXECUTABLE_SUFFIX}" DESTINATION bin/node)
else()
  install(PROGRAMS "${WHIPPET_NODE_DIR}/whippetd" "${WHIPPET_NODE_DIR}/whippet-cli" DESTINATION libexec/whippet)
endif()
install(FILES ../COPYING packaging/THIRD_PARTY.txt vendor/SHA256SUMS.txt DESTINATION share/whippet)
install(DIRECTORY packaging/licenses/ DESTINATION share/whippet/licenses)
if(WIN32)
  set(CMAKE_INSTALL_SYSTEM_RUNTIME_DESTINATION bin)
  include(InstallRequiredSystemLibraries)
endif()
if(WIN32 OR APPLE)
  qt_generate_deploy_app_script(TARGET whippet-desktop OUTPUT_SCRIPT deploy_script NO_UNSUPPORTED_PLATFORM_ERROR)
  install(SCRIPT ${deploy_script})
endif()
if(UNIX AND NOT APPLE)
  install(FILES packaging/whippet.desktop DESTINATION share/applications)
  install(FILES packaging/whippet.svg DESTINATION share/icons/hicolor/scalable/apps)
endif()
set(CPACK_PACKAGE_NAME Whippet)
set(CPACK_PACKAGE_VENDOR "Whippet Desktop")
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "Compact native Whippet wallet, miner and full node")
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
set(CPACK_PACKAGE_INSTALL_DIRECTORY Whippet)
set(CPACK_RESOURCE_FILE_LICENSE "${CMAKE_CURRENT_SOURCE_DIR}/../COPYING")
set(CPACK_PACKAGE_EXECUTABLES Whippet Whippet)
set(CPACK_DEBIAN_PACKAGE_MAINTAINER "Whippet Desktop maintainers")
set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
set(CPACK_DEBIAN_PACKAGE_DEPENDS "libxcb-cursor0, libxkbcommon-x11-0")
set(CPACK_DMG_VOLUME_NAME Whippet)
if(WIN32)
  set(CPACK_GENERATOR "ZIP")
elseif(APPLE)
  set(CPACK_GENERATOR "DragNDrop;TGZ")
else()
  set(CPACK_GENERATOR "DEB;TGZ")
  set(CPACK_PACKAGING_INSTALL_PREFIX "/usr")
endif()
include(CPack)
